cmake_minimum_required(VERSION 3.16)
project(rk3568_ov13850_robot LANGUAGES CXX)

option(ROBOT_BUILD_CAMERA "Build the Linux camera application" ON)
option(ROBOT_WITH_RKNN "Enable RKNN inference" OFF)
option(ROBOT_WITH_RGA "Enable RGA preprocessing" OFF)
option(ROBOT_WITH_MPP "Enable MPP encoding and RTMP publishing" OFF)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(robot_core OBJECT src/config.cpp src/nv12.cpp src/yolo.cpp)
target_include_directories(robot_core PUBLIC include)

include(CTest)
if(BUILD_TESTING OR ROBOT_BUILD_CAMERA)
    find_package(Threads REQUIRED)
endif()
if(BUILD_TESTING)
    add_executable(robot_core_test tests/core_test.cpp)
    target_link_libraries(robot_core_test PRIVATE robot_core Threads::Threads)
    add_test(NAME robot_core_test COMMAND robot_core_test "${CMAKE_CURRENT_SOURCE_DIR}/config/robot.conf")
endif()

if(NOT ROBOT_BUILD_CAMERA)
    return()
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The camera requires Linux. For host tests use -DROBOT_BUILD_CAMERA=OFF.")
endif()
if(ROBOT_WITH_RGA AND NOT ROBOT_WITH_RKNN)
    message(FATAL_ERROR "ROBOT_WITH_RGA requires ROBOT_WITH_RKNN.")
endif()

add_executable(inspection_robot src/main.cpp src/capture.cpp)
target_link_libraries(inspection_robot PRIVATE robot_core Threads::Threads)
target_compile_options(inspection_robot PRIVATE -Wall -Wextra -Wpedantic)
target_compile_options(robot_core PRIVATE -Wall -Wextra -Wpedantic)

if(ROBOT_WITH_RKNN)
    find_path(RKNN_INCLUDE_DIR rknn_api.h PATH_SUFFIXES rknn rknpu2)
    find_library(RKNN_LIBRARY NAMES rknnrt)
    if(NOT RKNN_INCLUDE_DIR OR NOT RKNN_LIBRARY)
        message(FATAL_ERROR "Set RKNN_INCLUDE_DIR and RKNN_LIBRARY from the RK3568 SDK.")
    endif()
    target_include_directories(inspection_robot SYSTEM PRIVATE "${RKNN_INCLUDE_DIR}")
    target_link_libraries(inspection_robot PRIVATE "${RKNN_LIBRARY}")
    target_compile_definitions(inspection_robot PRIVATE ROBOT_WITH_RKNN)
    target_sources(inspection_robot PRIVATE src/detector.cpp)
endif()
if(ROBOT_WITH_RGA)
    find_path(RGA_INCLUDE_DIR im2d.h PATH_SUFFIXES rga)
    find_library(RGA_LIBRARY NAMES rga)
    if(NOT RGA_INCLUDE_DIR OR NOT RGA_LIBRARY)
        message(FATAL_ERROR "Set RGA_INCLUDE_DIR and RGA_LIBRARY from the RK3568 SDK.")
    endif()
    target_include_directories(inspection_robot SYSTEM PRIVATE "${RGA_INCLUDE_DIR}")
    target_link_libraries(inspection_robot PRIVATE "${RGA_LIBRARY}")
    target_compile_definitions(inspection_robot PRIVATE ROBOT_WITH_RGA)
endif()
if(ROBOT_WITH_MPP)
    find_path(MPP_INCLUDE_DIR rk_mpi.h PATH_SUFFIXES rockchip mpp)
    find_library(MPP_LIBRARY NAMES rockchip_mpp mpp)
    if(NOT MPP_INCLUDE_DIR OR NOT MPP_LIBRARY)
        message(FATAL_ERROR "Set MPP_INCLUDE_DIR and MPP_LIBRARY from the RK3568 SDK.")
    endif()
    target_include_directories(inspection_robot SYSTEM PRIVATE "${MPP_INCLUDE_DIR}")
    target_link_libraries(inspection_robot PRIVATE "${MPP_LIBRARY}")
    target_compile_definitions(inspection_robot PRIVATE ROBOT_WITH_MPP)
    target_sources(inspection_robot PRIVATE src/encoder.cpp src/rtmp.cpp)
endif()
